services:
  kafka:
    image: apache/kafka
    container_name: kafka
    ports:
      - "9092:9092"
    networks:
      - telemetry-net   
    environment:
      KAFKA_NODE_ID: 1
      KAFKA_PROCESS_ROLES: broker,controller
      KAFKA_LISTENERS: PLAINTEXT://:9092,CONTROLLER://:9093
      KAFKA_ADVERTISED_LISTENERS: PLAINTEXT://kafka:9092
      KAFKA_CONTROLLER_QUORUM_VOTERS: 1@kafka:9093
      KAFKA_CONTROLLER_LISTENER_NAMES: CONTROLLER
      KAFKA_OFFSETS_TOPIC_REPLICATION_FACTOR: 1
      KAFKA_TRANSACTION_STATE_LOG_REPLICATION_FACTOR: 1
      KAFKA_TRANSACTION_STATE_LOG_MIN_ISR: 1
    # AJOUT DU HEALTHCHECK 
    healthcheck:
      test: ["CMD-SHELL", "nc -z localhost 9092 || exit 1"]
      interval: 5s
      timeout: 10s
      retries: 10

  kafka-init:
    image: apache/kafka
    depends_on:
      kafka:
        condition: service_healthy # Attend que le healthcheck soit OK
    networks:
      - telemetry-net    
    entrypoint: [ "/bin/sh", "-c" ]
    command: |
      /opt/kafka/bin/kafka-topics.sh --create --topic telemetry --bootstrap-server kafka:9092 --if-not-exists
      /opt/kafka/bin/kafka-topics.sh --create --topic alerts --bootstrap-server kafka:9092 --if-not-exists

  producer:
    build: ./producer
    depends_on:
      kafka:
        condition: service_healthy # Synchronisation intelligente
    environment:
      KAFKA_BROKER: kafka:9092
    networks:
      - telemetry-net

  consumer:
    build: ./consumer
    depends_on:
      kafka:
        condition: service_healthy # Évite l'erreur Connection Refused
    environment:
      KAFKA_BROKER: kafka:9092
    networks:
      - telemetry-net

  api:
    build: ./api
    ports:
      - "5000:5000"
    networks:
      - telemetry-net

networks:
  telemetry-net:

docker compose up -d --build
